เชื่อมข้อมูลและราคาตลาดกับชีต "เงินป้าเล็ก"

ใน Google Sheet เงินป้าเล็ก ได้เพิ่มแท็บ พอร์ตทองคำและหุ้นรายไม้ ที่ผูกสูตร =GOOGLEFINANCE() สำหรับดึงราคาหุ้นและคำนวณทองคำทั้ง 3 ไม้ไว้ให้เรียบร้อยแล้ว

📜 โค้ดสำหรับวางในเมนู Extensions > Apps Script ของชีต เงินป้าเล็ก: 
function doGet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('ชีต1');
  const lastRow = Math.max(sheet.getLastRow(), 4);
  const rows = lastRow > 4 ? sheet.getRange(5, 6, lastRow - 4, 5).getValues() : [];
  const data = rows.filter(r => r[0] && r[1]).map(r => ({
    timestamp: Utilities.formatDate(new Date(r[0]), 'Asia/Bangkok', 'yyyy-MM-dd HH:mm:ss'),
    category: String(r[1]),
    detail: String(r[2] || ''),
    amount: Number(r[3] || 0),
    note: String(r[4] || '')
  }));
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok', transactions: data }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('ชีต1');
  const payload = JSON.parse(e.postData.contents);
  if (payload.action === 'replaceAll' && Array.isArray(payload.transactions)) {
    sheet.getRange('F5:J500').clearContent();
    if (payload.transactions.length > 0) {
      const values = payload.transactions.map(t => [t.timestamp, t.category, t.detail, Number(t.amount), t.note]);
      sheet.getRange(5, 6, values.length, 5).setValues(values);
    }
  } else if (payload.transaction) {
    const t = payload.transaction;
    const colF = sheet.getRange('F5:F500').getValues();
    let nextRow = 5;
    while (nextRow - 5 < colF.length && colF[nextRow - 5][0] !== '') nextRow++;
    sheet.getRange(nextRow, 6, 1, 5).setValues([[t.timestamp, t.category, t.detail, Number(t.amount), t.note]]);
  }
  return ContentService.createTextOutput(JSON.stringify({ status: 'ok' }))
    .setMimeType(ContentService.MimeType.JSON);
}
